# AutonomousTrust C-side conformance harness.
#
# Reads the JSON-mirrored corpus produced at build time by the Python
# preprocessor (`tools/corpus_to_json.py`), runs each case through a
# protocol-specific adapter, and emits a results JSON whose shape matches
# the Python harness's so the cross-language diff tool can compare them.

# Source paths.  `jcs.c` is also reused by `jcs_test`; both targets pull it
# in via source rather than a separate library, since at this scale a tiny
# duplicated compile keeps the CMake graph simpler than yet another lib.
# (`at_ryu` is defined at the top-level CMakeLists so both the conformance
# runner and the JCS parity test can link against it.)
set(AT_CONFORMANCE_SOURCES
    runner.c
    scenario_loader.c
    scenario_engine.c
    case_result.c
    negative_runner.c
    jcs.c
    adapters/network.c
    adapters/identity.c
    adapters/agreement.c
    adapters/negotiation.c
    adapters/reputation.c
    adapters/bootstrap.c
)

# The verification layers are extension libraries (FEATURE_SPLIT_PLAN Phase 3):
# each one's adapter is built, and the library linked whole, only when that
# library is. A protocol whose layer this build lacks then reads as "no C
# adapter" -- a skip, not a failure.
set(AT_CONFORMANCE_EXTENSIONS)
foreach(_pair physics:physics calibration:calibration prequential:prequential
              replication:replication certificates:certificate)
    string(REPLACE ":" ";" _pair ${_pair})
    list(GET _pair 0 _layer)
    list(GET _pair 1 _adapter)
    if(TARGET at_${_layer}_static)
        list(APPEND AT_CONFORMANCE_SOURCES adapters/${_adapter}.c)
        list(APPEND AT_CONFORMANCE_EXTENSIONS at_${_layer}_static)
        string(TOUPPER ${_layer} _up)
        list(APPEND AT_CONFORMANCE_DEFS AT_CONF_ORACLE_${_up})
    endif()
endforeach()
# ZTA (FEATURE_SPLIT_PLAN Phase 6): the identity and reputation adapters have a
# ZTA half, compiled in -- and libat_zta linked whole -- only when the library
# is built. Without it, a scenario with a zta_policy fixture skips.
if(TARGET at_zta_static)
    list(APPEND AT_CONFORMANCE_EXTENSIONS at_zta_static)
    list(APPEND AT_CONFORMANCE_DEFS AT_ZTA_ENABLED)
endif()
# First contact (FEATURE_SPLIT_PLAN Phase 7): its own protocol (contacts)
# registers through at_conformance_adapter below; the identity and negotiation
# adapters have a first-contact half, compiled in -- and the library linked
# whole -- only when it is built. Without it those scenarios skip.
# Rendezvous (Phase 7b): the network adapter's roster scenarios and the
# identity adapter's reachability records need it, and first contact links it.
if(TARGET at_rendezvous_static)
    list(APPEND AT_CONFORMANCE_EXTENSIONS at_rendezvous_static)
    list(APPEND AT_CONFORMANCE_DEFS AT_RENDEZVOUS_ENABLED)
endif()
if(TARGET at_first_contact_static)
    list(APPEND AT_CONFORMANCE_DEFS AT_FIRST_CONTACT_ENABLED)
endif()
# Adapters an extension registered with at_conformance_adapter()
# (extensions/at_extension.cmake), e.g. an external extension's own protocol.
# Their sources are compiled in, their libraries linked whole, and runner.c
# reaches them through the generated conformance_registry.c; a protocol no
# registered adapter owns reads as "no C adapter" (a skip).
get_property(_ext_adapters GLOBAL PROPERTY AT_CONF_EXT_ADAPTERS)
get_property(_ext_sources GLOBAL PROPERTY AT_CONF_EXT_SOURCES)
get_property(_ext_libs GLOBAL PROPERTY AT_CONF_EXT_LIBS)
get_property(_ext_includes GLOBAL PROPERTY AT_CONF_EXT_INCLUDES)
get_property(AT_CONFORMANCE_EXTRA_CORPORA GLOBAL PROPERTY AT_CONF_EXT_CORPORA)
list(APPEND AT_CONFORMANCE_SOURCES ${_ext_sources})
list(APPEND AT_CONFORMANCE_EXTENSIONS ${_ext_libs})
set(_reg_decls "")
set(_reg_rows "")
foreach(_pair ${_ext_adapters})
    string(REPLACE "=" ";" _pair "${_pair}")
    list(GET _pair 0 _proto)
    list(GET _pair 1 _fn)
    string(APPEND _reg_decls "void ${_fn}(const at_case_t *c, at_case_result_t *out);\n")
    string(APPEND _reg_rows "    {\"${_proto}\", ${_fn}},\n")
endforeach()
set(_reg_file ${CMAKE_CURRENT_BINARY_DIR}/conformance_registry.c)
file(CONFIGURE OUTPUT ${_reg_file} CONTENT
"/* Generated by src/c/conformance/CMakeLists.txt from at_conformance_adapter()
 * registrations. Do not edit. */
#include <stddef.h>
#include \"conformance_registry.h\"

${_reg_decls}
const at_conf_adapter_t at_conf_ext_adapters[] = {
${_reg_rows}    {NULL, NULL},
};
")
list(APPEND AT_CONFORMANCE_SOURCES ${_reg_file})

# The corpus YAML lives next to src/c/, under src/autonomous-trust/conformance/.
# CMake needs an absolute path to drive the YAML→JSON conversion via Python.
set(AT_CONFORMANCE_CORPUS_YAML
    ${CMAKE_SOURCE_DIR}/../autonomous-trust/conformance
    CACHE PATH "Path to the YAML conformance corpus root")

set(AT_CONFORMANCE_CORPUS_JSON
    ${CMAKE_BINARY_DIR}/conformance/corpus-json
    CACHE PATH "Build-time output dir for JSON-mirrored corpus")

set(AT_CONFORMANCE_RESULTS
    ${CMAKE_BINARY_DIR}/conformance/results
    CACHE PATH "Where conformance_c writes its results JSON")

find_package(Python3 COMPONENTS Interpreter REQUIRED)

# Custom command: run the YAML→JSON converter. Inputs are every YAML file
# under the corpus; output is the index.json (ninja/make rebuild key).
set(_corpus_globs
    ${AT_CONFORMANCE_CORPUS_YAML}/scenarios/*.yaml
    ${AT_CONFORMANCE_CORPUS_YAML}/vectors/*.yaml)
set(_extra_in)
foreach(_c ${AT_CONFORMANCE_EXTRA_CORPORA})
    list(APPEND _corpus_globs ${_c}/scenarios/*.yaml ${_c}/vectors/*.yaml)
    list(APPEND _extra_in --in ${_c})
endforeach()
file(GLOB_RECURSE AT_CONFORMANCE_YAML_FILES CONFIGURE_DEPENDS ${_corpus_globs})

add_custom_command(
    OUTPUT ${AT_CONFORMANCE_CORPUS_JSON}/index.json
    COMMAND ${CMAKE_COMMAND} -E env
            "PYTHONPATH=${CMAKE_SOURCE_DIR}/../autonomous-trust"
            ${Python3_EXECUTABLE} -m tools.corpus_to_json
            --in ${AT_CONFORMANCE_CORPUS_YAML} ${_extra_in}
            --out ${AT_CONFORMANCE_CORPUS_JSON}
    DEPENDS ${AT_CONFORMANCE_YAML_FILES}
            ${CMAKE_SOURCE_DIR}/../autonomous-trust/tools/corpus_to_json.py
            ${CMAKE_SOURCE_DIR}/../autonomous-trust/conformance/harness/common/scenario_loader.py
    COMMENT "Converting YAML corpus -> JSON for C harness"
    VERBATIM
)

add_custom_target(at_conformance_json
    DEPENDS ${AT_CONFORMANCE_CORPUS_JSON}/index.json
)

# The runner executable. Linked against the existing static lib to pick up
# any future C-side wire/scenario adapters that need framework symbols.
add_executable(conformance_runner ${AT_CONFORMANCE_SOURCES})
add_dependencies(conformance_runner at_conformance_json)

target_include_directories(conformance_runner
    PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}
        ${CMAKE_SOURCE_DIR}/autonomous_trust
        ${CMAKE_SOURCE_DIR}/extensions
        # an external adapter includes "identity_priv.h" and its extension's
        # own headers by name
        ${CMAKE_CURRENT_SOURCE_DIR}/adapters
        ${_ext_includes}
)
target_compile_definitions(conformance_runner PRIVATE ${AT_CONFORMANCE_DEFS})

# Mirror the test executables' link line: --start-group/--end-group around
# the static archive lets the linker make multiple resolution passes, which
# is necessary because msg_types.c (and the protobuf-c generated objects)
# pull in symbols defined later in the archive's link order. Without this,
# protobuf_c_message_* symbols come up undefined even though they're in
# libprotobuf-c, because libsymbols are resolved single-pass.
set(AT_CONFORMANCE_LIBS
    -Wl,--whole-archive ${AT_CONFORMANCE_EXTENSIONS} -Wl,--no-whole-archive
    -Wl,--start-group autonomous_trust_static ${TEST_DEPENDENCIES} -Wl,--end-group
    at_ryu
)
target_link_libraries(conformance_runner PRIVATE ${AT_CONFORMANCE_LIBS})
if(TARGET at_zta_static)
    target_link_libraries(conformance_runner PRIVATE OpenSSL::SSL OpenSSL::Crypto)
endif()

# A `make conformance_c` target: build, then run, writing results JSON.
add_custom_target(conformance_c
    COMMAND ${CMAKE_COMMAND} -E make_directory ${AT_CONFORMANCE_RESULTS}
    COMMAND ${CMAKE_COMMAND} -E echo "Running C conformance harness..."
    COMMAND $<TARGET_FILE:conformance_runner>
            ${AT_CONFORMANCE_CORPUS_JSON}
            ${AT_CONFORMANCE_RESULTS}/c-latest.json
    DEPENDS conformance_runner at_conformance_json
    USES_TERMINAL
)
